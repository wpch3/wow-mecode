## 执行一个完整原子块

本机`py -3`记录已失效，使用已确认存在的Python 3.12。下面代码必须整块执行；任何红字都会停止块内后续动作，不会再出现“前一步失败、后一步继续”的连锁假错误。

```powershell
& {
    $ErrorActionPreference = "Stop"
    $Python = "C:\Users\Administrator\AppData\Local\Programs\Python\Python312\python.exe"
    $Probe  = "$env:USERPROFILE\Downloads\probe_combo_source.py"
    $Out    = "$env:USERPROFILE\Desktop\actual_source_probe_$(Get-Date -Format 'yyyyMMdd_HHmmss').txt"
    $ExpectedProbe = "0b215db4be55f19b85d4112d7a77de0f9851371fff1219296cfcd6dd41969db8"

    foreach ($Path in @($Python, $Probe)) {
        if (-not (Test-Path -LiteralPath $Path -PathType Leaf)) {
            throw "找不到文件：$Path。请先把probe_combo_source.py下载到Downloads。"
        }
    }
    $ActualProbe = (Get-FileHash -LiteralPath $Probe -Algorithm SHA256).Hash.ToLowerInvariant()
    if ($ActualProbe -ne $ExpectedProbe) {
        throw "F44探针SHA不符：实际=$ActualProbe，预期=$ExpectedProbe"
    }

    & $Python $Probe --self-test
    if ($LASTEXITCODE -ne 0) { throw "F44探针self-test失败，退出码=$LASTEXITCODE" }

    & $Python $Probe --source-root "D:\TrinityCore" | Tee-Object -FilePath $Out
    if ($LASTEXITCODE -ne 0) { throw "F44真实源码探针失败，退出码=$LASTEXITCODE" }

    $Report = Get-Content -LiteralPath $Out -Raw
    foreach ($Marker in @(
        "READY_FOR_EXACT_FIX=True",
        "SOURCE_OR_CONFIG_EDITS=0",
        "DATABASE_EDITS=0"
    )) {
        if (-not $Report.Contains($Marker)) { throw "报告缺少标志：$Marker" }
    }
    Write-Host "[OK] F44真实源码只读探针完成"
    Write-Host "[OK] 报告路径=$Out"
}
```
