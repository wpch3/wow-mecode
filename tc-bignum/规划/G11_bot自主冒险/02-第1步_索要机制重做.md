## 先修一个隐藏的数据 bug

### `npcbot_bond` 主键少了 `bot_kind`

实查 `01c_建bond表_一步到位.sql:70`：

```sql
PRIMARY KEY (`bot_id`, `owner_guid`),
```

**没有 `bot_kind`！**

后果：NPCBot 的 entry（比如 70231）和 PlayerBot 的角色 guid
如果撞号，两个 bot 的羁绊数据会**串在一起**。

你的 pbot 角色 guid 可能是 2、3 这种小数字，
暂时不会和 70xxx 撞。但一旦有玩家 guid 到了 70231，就出问题。

### 修复 SQL

```sql
-- G11第1步：给 npcbot_bond 主键加上 bot_kind
-- 先删旧主键再建新的（MySQL 不能直接改主键）

ALTER TABLE `characters`.`npcbot_bond`
  DROP PRIMARY KEY,
  ADD PRIMARY KEY (`bot_id`, `bot_kind`, `owner_guid`);

-- 验证
SELECT
    COLUMN_NAME AS `主键字段`,
    SEQ_IN_INDEX AS `顺序`
FROM `information_schema`.`STATISTICS`
WHERE TABLE_SCHEMA='characters' AND TABLE_NAME='npcbot_bond' AND INDEX_NAME='PRIMARY'
ORDER BY SEQ_IN_INDEX;
-- 应该输出 3 行：bot_id / bot_kind / owner_guid
```

> 如果报错说主键不存在，说明表结构和文档不一致，把
> `SHOW CREATE TABLE characters.npcbot_bond;` 的结果发我。

### 同步改 RecordGift 的 SQL

`bot_companion.cpp` 里搜：

```cpp
        "INSERT INTO npcbot_bond (bot_id, owner_guid, bot_kind, gift_points, gifts_to_bot, "
```

这句的 `ON DUPLICATE KEY UPDATE` 现在才会按 `bot_kind` 正确区分。
**SQL 文本不用改**，改了主键就自动生效。

### 替换为

```cpp
    // G11第1步：不再要求组队。改为在附近找"羁绊最高"的玩家搭话。
    // 用 AnyPlayerInPositionRangeCheck 而不是 AnyPlayerInObjectRangeCheck，
    // 因为后者内部走 IsWithinDistInMap（含相位检查），pbot 相位可能不同步。
    std::list<Player*> nearby;
    Trinity::AnyPlayerInPositionRangeCheck pcheck(bot, 40.0f, true);
    Trinity::PlayerListSearcher<Trinity::AnyPlayerInPositionRangeCheck> searcher(bot, nearby, pcheck);
    Cell::VisitWorldObjects(bot, searcher, 45.0f);

    Player* owner = nullptr;
    int bestScore = -1;
    uint8 botLevel = bot->GetLevel();

    for (Player* p : nearby)
    {
        if (!p || p == bot || !p->IsInWorld() || !p->IsAlive() || p->IsInCombat())
            continue;
        if (p->GetSession() && p->GetSession()->PlayerDisconnected())
            continue;

        int lvlDiff = int(p->GetLevel()) - int(botLevel);
        if (lvlDiff < -15 || lvlDiff > 15)
            continue;

        uint8 bondLv = BotCompanionMgr::GetBondLevel(
            bot->GetGUID().GetCounter(), 1, p->GetGUID().GetCounter());

        int score = int(bondLv) * 100;
        if (score > bestScore)
        {
            bestScore = score;
            owner = p;
        }
    }

    if (!owner)
    {
        timer = 30000;
        return;
    }

    // 全是陌生人（羁绊0）时只有 30% 概率开口，避免骚扰路人
    if (bestScore == 0 && urand(0, 9) >= 3)
    {
        timer = 60000;
        return;
    }
```

## 改动三：满足检测实现（bot_companion.cpp）

### 在 `ReloadAll()` 之前插入

```cpp
// ---------------------------------------------------------------------------
//  G11第1步：查最近10分钟有没有未满足的索要
// ---------------------------------------------------------------------------
uint8 BotCompanionMgr::GetPendingRequest(uint32 botId, uint8 botKind, uint32 ownerGuid)
{
    uint32 since = uint32(GameTime::GetGameTime()) - 600;

    QueryResult res = CharacterDatabase.PQuery(
        "SELECT request_type FROM npcbot_request_log WHERE bot_id = {} AND bot_kind = {} "
        "AND owner_guid = {} AND answered = 0 AND ask_time > {} ORDER BY id DESC LIMIT 1",
        botId, uint32(botKind), ownerGuid, since);

    return res ? res->Fetch()[0].GetUInt8() : 0;
}

// ---------------------------------------------------------------------------
//  G11第1步：你给的东西对不对口
// ---------------------------------------------------------------------------
bool BotCompanionMgr::DoesGiftMatchRequest(uint8 giftKind, uint8 reqType)
{
    switch (reqType)
    {
        case BOT_REQUEST_FOOD:    return giftKind == BOT_GIFT_KIND_FOOD;
        case BOT_REQUEST_DRINK:   return giftKind == BOT_GIFT_KIND_DRINK;
        case BOT_REQUEST_BANDAGE: return giftKind == BOT_GIFT_KIND_POTION || giftKind == BOT_GIFT_KIND_FOOD;
        case BOT_REQUEST_REPAIR:  return giftKind == BOT_GIFT_KIND_EQUIP;
        default:                  return false;
    }
}

// ---------------------------------------------------------------------------
//  G11第1步：标记满足 + 额外加分
// ---------------------------------------------------------------------------
void BotCompanionMgr::MarkRequestMet(uint32 botId, uint8 botKind, uint32 ownerGuid, uint8 reqType)
{
    uint32 now = uint32(GameTime::GetGameTime());
    uint32 since = now - 600;

    CharacterDatabase.PExecute(
        "UPDATE npcbot_request_log SET answered = 1, answer_time = {} "
        "WHERE bot_id = {} AND bot_kind = {} AND owner_guid = {} AND request_type = {} "
        "AND answered = 0 AND ask_time > {}",
        now, botId, uint32(botKind), ownerGuid, uint32(reqType), since);

    CharacterDatabase.PExecute(
        "UPDATE npcbot_bond SET requests_met = requests_met + 1, "
        "gift_points = gift_points + 15 "
        "WHERE bot_id = {} AND bot_kind = {} AND owner_guid = {}",
        botId, uint32(botKind), ownerGuid);
}

```

## 为什么用 `AnyPlayerInPositionRangeCheck`

```cpp
// GridNotifiers.h:1521  AnyPlayerInObjectRangeCheck
if (!_obj->IsWithinDistInMap(u, _range))     // <- 含 InSamePhase 检查

// GridNotifiers.h:1542  AnyPlayerInPositionRangeCheck
if (!u->IsWithinDist3d(_pos, _range))        // <- 纯距离
```

F26 我们踩过这个坑：pbot 用 `.pbot spawn` 上线**不同步相位**，
`IsWithinDistInMap` 会永远返回 false。

**用 `AnyPlayerInPositionRangeCheck` 绕开。**
