### 3.2 PickText 整体替换

搜索 `std::string BotCompanionMgr::PickText`，**整个函数替换为**：

```cpp
// G19第3步：场景感知 + 防重复
//   评分：地图+200 区域+150 场景+80 种族+100 职业+50
//   （地图/区域压过种族，因为"在这个副本里"比"我是亡灵"更具体）
//   冷却：说过的台词在 cooldown 秒内不再选，逼它把所有台词轮一遍
std::string BotCompanionMgr::PickText(uint8 careType, uint8 botClass, uint8 botRace/* = 0*/,
                                      uint8 scene/* = 0*/, uint32 zoneId/* = 0*/, uint32 mapId/* = 0*/,
                                      uint32 botGuid/* = 0*/)
{
    auto itr = _texts.find(careType);
    if (itr == _texts.end() || itr->second.empty())
        return std::string();

    time_t const now = GameTime::GetGameTime();
    auto& cdMap = _textCooldowns[botGuid];

    int bestScore = -1;
    std::vector<CompanionText const*> pool;

    for (CompanionText const& t : itr->second)
    {
        // ---- 硬性过滤：指定了就必须对上 ----
        if (t.BotClass != 0 && t.BotClass != botClass) continue;
        if (t.BotRace  != 0 && t.BotRace  != botRace)  continue;
        if (t.Scene    != 0 && t.Scene    != scene)    continue;
        if (t.ZoneId   != 0 && t.ZoneId   != zoneId)   continue;
        if (t.MapId    != 0 && t.MapId    != mapId)    continue;

        // ---- 冷却中的跳过（这是"所有台词都能出现"的关键）----
        if (botGuid)
        {
            auto cdItr = cdMap.find(t.Id);
            if (cdItr != cdMap.end() && cdItr->second > now)
                continue;
        }

        // ---- 打分 ----
        int score = 0;
        if (t.MapId    != 0) score += 200;   // 精确地图最优先
        if (t.ZoneId   != 0) score += 150;
        if (t.BotRace  != 0) score += 100;
        if (t.Scene    != 0) score += 80;
        if (t.BotClass != 0) score += 50;

        if (score > bestScore)
        {
            bestScore = score;
            pool.clear();
            pool.push_back(&t);
        }
        else if (score == bestScore)
            pool.push_back(&t);
    }

    // 全在冷却中 -> 清掉这个bot的冷却重来（保证不会哑巴）
    if (pool.empty() && botGuid && !cdMap.empty())
    {
        cdMap.clear();
        return std::string();      // 这次不说，下次就有的说了
    }

    if (pool.empty())
        return std::string();

    // 权重随机
    uint32 total = 0;
    for (CompanionText const* t : pool)
        total += t->Weight;

    CompanionText const* chosen = pool[0];
    if (total)
    {
        uint32 roll = urand(0, total - 1);
        for (CompanionText const* t : pool)
        {
            if (roll < t->Weight) { chosen = t; break; }
            roll -= t->Weight;
        }
    }

    // 登记冷却
    if (botGuid && chosen->Cooldown)
        cdMap[chosen->Id] = now + chosen->Cooldown;

    return chosen->Text;
}
```

## 步骤4 · `bot_ai.cpp` 算场景

### 4.1 加一个辅助函数

// G19第3步：算出 bot 当前所处的场景类别
//   1=副本 2=团本 3=主城 4=小镇 5=野外 6=战场 7=圣所
uint8 bot_ai::GetCurrentScene() const
{
    Map const* map = me->FindMap();          // Object.h:468 无断言版
    if (!map)
        return 0;

    if (map->IsBattlegroundOrArena())
        return 6;
    if (map->IsRaid())
        return 2;
    if (map->IsDungeon())
        return 1;

    // 非副本：看区域标志
    if (AreaTableEntry const* area = sAreaTableStore.LookupEntry(me->GetAreaId()))
    {
        if (area->Flags & AREA_FLAG_CAPITAL)    return 3;   // DBCEnums.h:255
        if (area->Flags & AREA_FLAG_SANCTUARY)  return 7;   // DBCEnums.h:258
        if (area->Flags & AREA_FLAG_TOWN)       return 4;   // DBCEnums.h:268
    }
    return 5;   // 野外
}

### 三道闸总览

| 闸 | 参数 | 效果 |
|---|---|---|
| 单条冷却 | `cooldown` 字段默认 900 秒 | 同一句话 15 分钟内不重复 |
| 全局间隔 | `_careChatTimer` 3-6 分钟 | bot 不会连珠炮 |
| 场景匹配 | `scene` 过滤 | 在副本里不说野外的话 |
