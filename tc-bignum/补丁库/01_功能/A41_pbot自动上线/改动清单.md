## 零、设计要点（为什么这么做）

### 为什么不能在 OnLogin 里直接 spawn

```
ScriptMgr.h:692   virtual void OnLogin(Player* player, bool firstLogin);   public(625段)
```

OnLogin 触发时，**你自己的角色还在登录流程中**。此时立刻 spawn bot 会有两个问题：

1. **竞态**：bot 登录也是异步的（`cs_playerbot.cpp:296 QueuePacket`），
   两个登录流程交叉，容易出状态错乱
2. **组队恢复未完成**：`Player.cpp:17317 _LoadGroup` 虽然早于 OnLogin，
   但队伍成员的 Player 对象未必都已就绪

**所以：OnLogin 只做一件事 —— 往延迟队列里记一笔。**
真正的 spawn 在 `OnUpdate` 里延迟 5 秒执行。

### 为什么要建表

`g_pbots` 是**内存变量**，服务器一重启就空了。
要实现"我上线，我的 bot 也上线"，必须把"我带了哪些 bot"存进数据库。

## 步骤1 · SQL（DBeaver 执行）

```sql
-- A41: PlayerBot 名册。记录"哪个玩家带了哪些 pbot"
CREATE TABLE IF NOT EXISTS `characters`.`playerbot_roster` (
    `owner_guid`  INT UNSIGNED NOT NULL           COMMENT '主人角色guid',
    `bot_account` INT UNSIGNED NOT NULL           COMMENT 'bot账号id',
    `bot_guid`    INT UNSIGNED NOT NULL           COMMENT 'bot角色guid',
    `bot_name`    VARCHAR(12)  NOT NULL DEFAULT '' COMMENT 'bot角色名',
    `auto_flags`  INT UNSIGNED NOT NULL DEFAULT 91 COMMENT '自动接受开关位',
    `enabled`     TINYINT UNSIGNED NOT NULL DEFAULT 1 COMMENT '1=主人上线时自动拉起',
    PRIMARY KEY (`owner_guid`, `bot_guid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
  COMMENT='A41 pbot自动上线名册';
```

> `auto_flags` 默认 91 = `PBOT_AUTO_DEFAULT`
> （GROUP 1 + GUILD 2 + RESURRECT 8 + SUMMON 16 + RELEASE 64 = 91）

**主键是 `(owner_guid, bot_guid)`** —— 同一个 bot 不能被同一个主人登记两次，
但不同主人可以各自登记（虽然实际同时只能上线一个）。

```cpp
// ============================================================================
//  A41 · spawn 核心（不依赖 ChatHandler，供自动上线复用）
// ============================================================================
//  原 SpawnBot 把"查找+校验+登录"和"给玩家发消息"混在一起，
//  自动上线时没有 ChatHandler，所以必须把核心逻辑抽出来。
//  这里是原 :252-300 那段的无 UI 版本。
// ============================================================================
bool PBotSpawnCore(uint32 accountId, ObjectGuid charGuid,
                   std::string const& charName, uint32 autoFlags,
                   std::string* reason)
{
    // 已经在线
    if (ObjectAccessor::FindPlayer(charGuid))
    {
        if (reason) *reason = "已经在线";
        return false;
    }

    // 账号已有会话
    if (sWorld->FindSession(accountId))
    {
        if (reason) *reason = "账号已有活动会话";
        return false;
    }

    // 造无连接会话（同 SpawnBot :263）
    std::string sessName = charName;
    WorldSession* sess = new WorldSession(
        accountId,
        std::move(sessName),
        nullptr,                    // 无 socket
        SEC_ADMINISTRATOR,
        2,                          // WotLK
        0,
        Minutes(0),
        LOCALE_zhCN,
        0,
        false);

    sess->SetBotSession(true);
    sess->AddLegitCharacter(charGuid);
    sWorld->AddSession(sess);

    WorldPacket* pkt = new WorldPacket(CMSG_PLAYER_LOGIN, 8);
    *pkt << charGuid;
    sess->QueuePacket(pkt);

    // 登记进内存表（去重：同一个 guid 只留一份）
    for (PBotEntry const& e : g_pbots)
        if (e.CharGuid == charGuid)
            return true;

    g_pbots.push_back({ accountId, charGuid, charName, autoFlags });
    return true;
}

// ============================================================================
//  A41 · 名册读写
// ============================================================================
void PBotLoadRoster(ObjectGuid ownerGuid, std::vector<PBotEntry>& out)
{
    out.clear();

    QueryResult res = CharacterDatabase.PQuery(
        "SELECT bot_account, bot_guid, bot_name, auto_flags "
        "FROM playerbot_roster WHERE owner_guid = {} AND enabled = 1",
        ownerGuid.GetCounter());

    if (!res)
        return;

    do
    {
        Field* f = res->Fetch();
        PBotEntry e;
        e.AccountId = f[0].GetUInt32();
        e.CharGuid  = ObjectGuid::Create<HighGuid::Player>(f[1].GetUInt32());
        e.CharName  = f[2].GetString();
        e.AutoFlags = f[3].GetUInt32();
        out.push_back(e);
    } while (res->NextRow());
}

void PBotRosterAdd(ObjectGuid ownerGuid, PBotEntry const& e)
{
    CharacterDatabase.PExecute(
        "REPLACE INTO playerbot_roster "
        "(owner_guid, bot_account, bot_guid, bot_name, auto_flags, enabled) "
        "VALUES ({}, {}, {}, '{}', {}, 1)",
        ownerGuid.GetCounter(), e.AccountId, e.CharGuid.GetCounter(),
        e.CharName, e.AutoFlags);
}

void PBotRosterRemove(ObjectGuid ownerGuid, ObjectGuid botGuid)
{
    CharacterDatabase.PExecute(
        "DELETE FROM playerbot_roster WHERE owner_guid = {} AND bot_guid = {}",
        ownerGuid.GetCounter(), botGuid.GetCounter());
}
```

```cpp
// ------------------------------------------------------------------
//  A41: 处理自动上线延迟队列
// ------------------------------------------------------------------
static void ProcessLoginQueue(uint32 diff)
{
    if (_pbotLoginQueue.empty())
        return;

    for (auto it = _pbotLoginQueue.begin(); it != _pbotLoginQueue.end(); )
    {
        // 倒计时
        if (it->second > diff)
        {
            it->second -= diff;
            ++it;
            continue;
        }

        // 到点了，取出数据
        ObjectGuid botGuid = it->first;
        auto dataItr = _pbotLoginData.find(botGuid);
        if (dataItr == _pbotLoginData.end())
        {
            // 数据丢了（不该发生），清掉这条继续
            it = _pbotLoginQueue.erase(it);
            continue;
        }

        // 【重要】先按值拷贝，再 erase。
        // erase 之后 dataItr 失效，不能再用 dataItr->second。
        PBotEntry const e = dataItr->second;

        // 先删记录再执行，避免失败时反复重试
        it = _pbotLoginQueue.erase(it);
        _pbotLoginData.erase(dataItr);

        std::string reason;
        if (PBotSpawnCore(e.AccountId, e.CharGuid, e.CharName, e.AutoFlags, &reason))
            TC_LOG_INFO("misc", "A41: 自动拉起 pbot {}", e.CharName);
        else
            TC_LOG_INFO("misc", "A41: pbot {} 拉起失败：{}", e.CharName, reason);
    }
}

```

### 4.4 加 OnLogin 钩子

在文件**末尾**，`AddSC_` 函数**上面**，插入：

```cpp
// ============================================================================
//  A41 · 主人上线 -> 排队拉起他的 pbot
// ============================================================================
class pbot_autologin_playerscript : public PlayerScript
{
public:
    pbot_autologin_playerscript() : PlayerScript("pbot_autologin_playerscript") { }

    // ScriptMgr.h:692  virtual void OnLogin(Player* player, bool firstLogin);  public(625段)
    void OnLogin(Player* player, bool /*firstLogin*/) override
    {
        if (!player)
            return;

        // bot 自己上线时不要再触发（否则套娃）
        if (player->GetSession() && player->GetSession()->PlayerDisconnected())
            return;

        std::vector<PBotEntry> roster;
        PBotLoadRoster(player->GetGUID(), roster);
        if (roster.empty())
            return;

        uint32 delay = 5000;    // 第一个 5 秒后拉起
        for (PBotEntry const& e : roster)
        {
            // 已经在线的跳过
            if (ObjectAccessor::FindPlayer(e.CharGuid))
                continue;

            _pbotLoginQueue[e.CharGuid] = delay;
            _pbotLoginData[e.CharGuid]  = e;
            delay += 2000;      // 每个间隔 2 秒，避免同时登录打爆
        }

        if (!_pbotLoginQueue.empty())
            ChatHandler(player->GetSession()).PSendSysMessage(
                "|cff00ff00[PlayerBot]|r 正在拉起 %u 个伙伴...",
                uint32(_pbotLoginQueue.size()));
    }
};

```

## 排错

### 如果没看到「正在拉起」

说明 `OnLogin` 没触发或名册是空的。DBeaver 查：

```sql
SELECT * FROM `characters`.`playerbot_roster`;
```

- **表是空的** → 步骤3.3 没装上，`.pbot spawn` 时没写库
- **表有数据但没提示** → 步骤4.5 没注册脚本

### 如果看到提示但 bot 没上线

看 `Server.log` 搜 `A41:`，会打印失败原因（`Logger.root=4` Warn，
`TC_LOG_INFO` 是 Info(3) **不会输出**）。

**要看这条日志，把 `TC_LOG_INFO` 改成 `TC_LOG_ERROR`**，或在 conf 里加：

```
Logger.misc=3,Console Server
```
